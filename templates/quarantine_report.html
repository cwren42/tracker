# QUARANTINE REPORT — Block Threat Sender Domains
# Generated: {{ now.strftime('%Y-%m-%d %H:%M') }} UTC  |  Period: Last {{ days }} days
# Source: {{ phish_malware_domains | length }} phishing/malware domains from quarantine data
# ─────────────────────────────────────────────────────────────
# Prerequisites: ExchangeOnlineManagement module v3+
# Requires role: Security Administrator or Exchange Administrator
# ─────────────────────────────────────────────────────────────

Connect-IPPSSession -UserPrincipalName admin@cirque.com

# Block phishing/malware sender domains in Tenant Allow/Block List
# These domains sent {{ phish_malware_domains | length }} messages classified as Phish or Malware
New-TenantAllowBlockListItems `
    -ListType Sender `
    -Block `
    -Entries @(
{% for d in phish_malware_domains %}        "{{ d }}"{{ "," if not loop.last }}
{% endfor %}    ) `
    -NoExpiration `
    -Notes "Phish/Malware campaign — blocked {{ now.strftime('%Y-%m-%d') }} by SOC ({{ total_in_period }} msgs in {{ days }}d)"

# Verify the entries were added
Get-TenantAllowBlockListItems -ListType Sender -Block | `
    Where-Object { $_.Value -in @({{ phish_malware_domains | map('tojson') | join(', ') }}) } | `
    Select-Object Value, ExpirationDate, Notes
Creates transport rules for persistent blocking with custom rejection messages. Use alongside TABL for defence-in-depth. 
# ─────────────────────────────────────────────────────────────
# QUARANTINE REPORT — Create Transport Rule Blocks
# Generated: {{ now.strftime('%Y-%m-%d %H:%M') }} UTC  |  Period: Last {{ days }} days
# ─────────────────────────────────────────────────────────────

Connect-ExchangeOnline -UserPrincipalName admin@cirque.com

# Block top phishing/malware domains with hard-reject transport rule
New-TransportRule `
    -Name "SECURITY: Block Known Phish/Malware Senders — SOC {{ now.strftime('%Y%m%d') }}" `
    -Enabled $true `
    -SenderDomainIs @(
{% for d in phish_malware_domains %}        "{{ d }}"{{ "," if not loop.last }}
{% endfor %}    ) `
    -RejectMessageReasonText "This message was blocked by Cirque Corporation security policy. Contact helpdesk@cirque.com if you believe this is an error." `
    -RejectMessageEnhancedStatusCode "5.7.1" `
    -Mode Enforce `
    -Comments "Created by SOC {{ now.strftime('%Y-%m-%d') }} — {{ total_in_period }} quarantined messages in {{ days }}d period"

# Review existing transport rules (look for conflicts)
Get-TransportRule | Where-Object { $_.Name -like "*SECURITY*" } | `
    Select-Object Name, Priority, Enabled, Mode | Sort-Object Priority
